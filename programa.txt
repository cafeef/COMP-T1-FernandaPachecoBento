print("Inicio do programa");

x = 2 + 5;
y = 10;
z = 0;

print("Valores iniciais:");
print(x);
print(y);
print(z);

print("Teste do operador AND:");
x < y and y > 5;

print("Teste do operador OR:");
x > y or y == 10;

print("Teste combinando atribuicao, comparacao e logica:");
z = x < y and y == 10;
print(z);

print("Teste do comando FOR:");
for(i = 0; i < 5; i = i + 1) print(i);

print("Teste do FOR com acumulador:");
soma = 0;
for(i = 1; i <= 5; i = i + 1) soma = soma + i;
print(soma);

print("Teste do IF com AND:");
if soma > 10 and x < y then print("soma maior que 10 e x menor que y");

print("Teste do WHILE com OR:");
contador = 0;

print(contador);
